# URL da API do Banco Central do Brasil
urlIPCA <- "https://api.bcb.gov.br/dados/serie/bcdata.sgs.433/dados?formato=json"
urlPTAX <- "https://api.bcb.gov.br/dados/serie/bcdata.sgs.3696/dados?formato=json"
urlSELIC <- "https://api.bcb.gov.br/dados/serie/bcdata.sgs.4189/dados?formato=json"

# Fazer a requisição GET
responseIPCA <- GET(urlIPCA)
responsePTAX <- GET(urlPTAX)
responseSELIC <- GET(urlSELIC)

dataIPCA <- content(responseIPCA, "text")
dataIPCA <- fromJSON(dataIPCA)

dataPTAX <- content(responsePTAX, "text")
dataPTAX <- fromJSON(dataPTAX)

dataSELIC <- content(responseSELIC, "text")
dataSELIC <- fromJSON(dataSELIC)


dataAll <- merge(dataIPCA,dataPTAX, by = "data", all = TRUE)
dataAll <- merge(dataAll, dataSELIC, by = "data", all = TRUE)
dataAll <- na.omit(dataAll)

colnames(dataAll) <- c("data", "IPCA", "PTAX", "SELIC")

dataAll[['data']] <- as.POSIXct(dataAll[['data']], format = "%d/%m/%Y")
dataAll <- dataAll[order(as.Date(dataAll$data, format="%d/%m/%Y")), ] 

dataAllFilter <- dataAll %>% filter(data >= '2000-01-01')

write.csv(dataAllFilter, "D:\\Afranc\\IPCA\\dadosR.csv")

dados <- data.frame(
  IPCA = as.numeric(dataAllFilter$IPCA),
  PTAX = as.numeric(dataAllFilter$PTAX),
  SELIC = as.numeric(dataAllFilter$SELIC)
)

# Ajustando o modelo de regressão linear múltipla com a nova variável
modelo <- lm(IPCA ~ PTAX+SELIC, data = dados)

# Resumo do modelo
a=summary(modelo)
print(a)
